-- Tablas para el sistema de Ajedrez

-- Clasificación de personajes (modelos GLTF)
CREATE TABLE IF NOT EXISTS chess_characters (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    gltf_path TEXT NOT NULL,
    piece_type TEXT NOT NULL, -- Rey, Reina, Torre, Alfil, Caballo, Peon
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Selecciones de los usuarios
CREATE TABLE IF NOT EXISTS chess_selections (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES usuarios(id) ON DELETE CASCADE,
    piece_type TEXT NOT NULL,
    character_id UUID REFERENCES chess_characters(id) ON DELETE CASCADE,
    is_opponent BOOLEAN DEFAULT FALSE, -- FALSE para blancas (jugador), TRUE para negras (contrincante)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(user_id, piece_type, is_opponent)
);

-- Asegurar que la tabla usuarios tenga la columna role
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='usuarios' AND column_name='role') THEN
        ALTER TABLE usuarios ADD COLUMN role TEXT DEFAULT 'user';
    END IF;
END $$;
