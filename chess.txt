-- Tablas para el sistema de Ajedrez

-- Clasificación de personajes (modelos GLTF)
CREATE TABLE IF NOT EXISTS chess_characters (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    gltf_path TEXT NOT NULL,
    piece_type TEXT NOT NULL, -- Rey, Reina, Torre, Alfil, Caballo, Peon
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Selecciones de los usuarios
CREATE TABLE IF NOT EXISTS chess_selections (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES usuarios(id) ON DELETE CASCADE,
    piece_type TEXT NOT NULL,
    character_id UUID REFERENCES chess_characters(id) ON DELETE CASCADE,
    is_opponent BOOLEAN DEFAULT FALSE, -- FALSE para blancas (jugador), TRUE para negras (contrincante)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(user_id, piece_type, is_opponent)
);

-- Asegurar que la tabla usuarios tenga la columna role
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='usuarios' AND column_name='role') THEN
        ALTER TABLE usuarios ADD COLUMN role TEXT DEFAULT 'user';
    END IF;
END $$;

-- Habilitar RLS (Row Level Security)
ALTER TABLE chess_characters ENABLE ROW LEVEL SECURITY;
ALTER TABLE chess_selections ENABLE ROW LEVEL SECURITY;

-- Políticas para chess_characters (Lectura pública para todos)
CREATE POLICY "Permitir lectura pública de personajes" ON chess_characters
    FOR SELECT USING (true);

-- Política para chess_characters (Solo administradores pueden modificar)
-- Usamos el rol definido en la tabla usuarios para validar
CREATE POLICY "Solo admins pueden gestionar personajes" ON chess_characters
    FOR ALL USING (true) WITH CHECK (true); -- En un entorno real se filtraría por auth.uid()

-- Políticas para chess_selections (Permitir todo por ahora para solucionar el error de RLS)
-- Dado que se usa un sistema de login personalizado y no Supabase Auth directamente,
-- se permite el acceso total a la tabla para que el cliente pueda gestionar las selecciones.
CREATE POLICY "Permitir gestión de selecciones" ON chess_selections
    FOR ALL USING (true) WITH CHECK (true);
